app_code = r'''
import numpy as np
import pandas as pd
import streamlit as st

from sklearn.preprocessing import MinMaxScaler
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense, Dropout, Input
from tensorflow.keras.callbacks import EarlyStopping

st.set_page_config(page_title="Electricity Forecast", page_icon="⚡", layout="wide")


@st.cache_data
def create_dataset():
    np.random.seed(42)
    dates = pd.date_range("2023-01-01", "2025-12-31", freq="D")
    day = np.arange(len(dates))
    weekly = 15 * np.sin(2 * np.pi * day / 7)
    yearly = 12 * np.sin(2 * np.pi * day / 365)
    noise = np.random.normal(0, 2, len(dates))
    consumption = 120 + weekly + yearly + noise
    price = 8 + 0.03 * (consumption - 120) + np.random.normal(0, 0.15, len(dates))
    price = np.maximum(price, 5)
    return pd.DataFrame({
        "Date": dates,
        "Consumption": consumption,
        "Price_per_kWh": price,
    })


@st.cache_resource(show_spinner=False)
def train_model():
    df = create_dataset()
    scaler = MinMaxScaler()
    scaled = scaler.fit_transform(df[["Consumption"]]).flatten()

    LOOKBACK, FORECAST_DAYS = 30, 30
    X, y = [], []
    for i in range(len(scaled) - LOOKBACK - FORECAST_DAYS + 1):
        X.append(scaled[i:i + LOOKBACK])
        y.append(scaled[i + LOOKBACK:i + LOOKBACK + FORECAST_DAYS])
    X = np.array(X).reshape(-1, LOOKBACK, 1)
    y = np.array(y)
    split = int(len(X) * 0.80)

    model = Sequential([
        Input(shape=(LOOKBACK, 1)),
        LSTM(64, return_sequences=True),
        Dropout(0.10),
        LSTM(32),
        Dropout(0.10),
        Dense(32, activation="relu"),
        Dense(FORECAST_DAYS),
    ])
    model.compile(optimizer="adam", loss="mse")
    early_stop = EarlyStopping(monitor="val_loss", patience=5, restore_best_weights=True)
    model.fit(X[:split], y[:split], epochs=40, batch_size=32,
              validation_split=0.10, callbacks=[early_stop], verbose=0)
    return model, scaler, scaled


st.title("⚡ Electricity Consumption and Price Forecasting")
st.write("Enter your electricity details to generate the forecast and estimated monthly bill.")

st.sidebar.header("Electricity Input")

consumer_type = st.sidebar.selectbox("Consumer Type", ["Household", "Industrial"])

month_names = [
    "January", "February", "March", "April", "May", "June",
    "July", "August", "September", "October", "November", "December",
]
selected_month = st.sidebar.selectbox("Forecast Month", month_names, index=1)

unit_cost = st.sidebar.number_input(
    "Electricity Unit Cost (₹/kWh)",
    min_value=1.0, max_value=100.0, value=8.0, step=0.5,
)

if consumer_type == "Household":
    number_of_people = st.sidebar.number_input(
        "Number of People", min_value=1, max_value=20, value=4, step=1
    )
    daily_consumption = st.sidebar.number_input(
        "Average Daily Consumption (kWh)",
        min_value=5.0, max_value=100.0, value=8.0, step=1.0,
    )
else:
    industry_size = st.sidebar.selectbox(
        "Industry Size", ["Small Industry", "Medium Industry", "Large Industry"]
    )
    limits = {
        "Small Industry": (50, 1000, 250),
        "Medium Industry": (500, 5000, 1000),
        "Large Industry": (2000, 20000, 5000),
    }
    minimum, maximum, default = limits[industry_size]
    daily_consumption = st.sidebar.number_input(
        "Average Daily Consumption (kWh)",
        min_value=float(minimum), max_value=float(maximum),
        value=float(default), step=50.0,
    )

st.divider()

if st.button("🚀 Generate Electricity Forecast", type="primary", use_container_width=True):
    model, scaler, scaled = train_model()

    LOOKBACK = 30
    seq = scaled[-LOOKBACK:].reshape(1, LOOKBACK, 1)
    preds = scaler.inverse_transform(
        model.predict(seq, verbose=0)[0].reshape(-1, 1)
    ).flatten()

    dates = pd.date_range(pd.Timestamp.today().normalize() + pd.Timedelta(days=1), periods=30, freq="D")

    daily_bill = preds * unit_cost
    total_consumption = preds.sum()
    total_bill = daily_bill.sum()
    average_consumption = preds.mean()

    forecast_df = pd.DataFrame({
        "Date": dates,
        "Predicted Consumption (kWh)": np.round(preds, 2),
        "Unit Cost (₹/kWh)": unit_cost,
        "Estimated Daily Bill (₹)": np.round(daily_bill, 2),
    })

    st.divider()
    st.subheader("30 Day Electricity Forecast")
    m1, m2, m3 = st.columns(3)
    m1.metric("Total Consumption", f"{total_consumption:,.2f} kWh")
    m2.metric("Average Daily Consumption", f"{average_consumption:,.2f} kWh")
    m3.metric("Estimated Monthly Bill", f"₹{total_bill:,.2f}")

    st.divider()
    st.subheader("Daily Forecast")
    st.dataframe(forecast_df, use_container_width=True, hide_index=True)
'''


with open("/content/app.py", "w", encoding="utf-8") as f:
    f.write(app_code)
